cURL Request
curl -X POST https://api.f-motion.com/v1/integrations/project-imports \
  -H "Authorization: Bearer fm_live_…" \
  -H "Content-Type: application/json" \
  -d '{
    "client_reference": "usr_99812",
    "media_assets": [
      { "type": "video", "url": "s3://bucket/raw_1.mp4" },
      { "type": "audio", "url": "https://host.com/track.wav" }
    ],
    "template_id": "tmpl_cinematic_01"
  }'
